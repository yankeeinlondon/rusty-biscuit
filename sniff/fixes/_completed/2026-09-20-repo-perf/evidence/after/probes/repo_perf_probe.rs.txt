//! Disposable Phase 4 probe for `2026-09-20-repo-perf` (not committed).
//!
//! Modes:
//! - `counters <root>`: one `detect_repo_structure` under a fresh collector.
//! - `tiny-latency`: single-request latency on a tiny tree, cases alternated.
//! - `concurrent <tiny|path> <threads> <requests-per-thread>`: concurrent
//!   `detect_repo_structure` calls; throughput and per-request latency.
//!
//! Build with `--features bench-internals`.

use std::path::{Path, PathBuf};
use std::sync::{Arc, Barrier};
use std::time::{Duration, Instant};

use sniff::filesystem::repo::{detect_repo_structure, nested_benchmark};
use sniff::performance::{PerformanceCollector, with_current_collector};

fn tiny_tree() -> PathBuf {
    let root = std::env::temp_dir().join(format!("repo-perf-tiny-{}", std::process::id()));
    let _ = std::fs::remove_dir_all(&root);
    let files: &[(&str, &str)] = &[
        (
            "Cargo.toml",
            "[workspace]\nmembers = [\"crates/a\", \"crates/b\"]\nresolver = \"2\"\n",
        ),
        ("README.md", "# tiny\n"),
        (
            "crates/a/Cargo.toml",
            "[package]\nname = \"a\"\nversion = \"0.1.0\"\nedition = \"2021\"\n",
        ),
        ("crates/a/src/lib.rs", "pub fn a() {}\n"),
        (
            "crates/b/Cargo.toml",
            "[package]\nname = \"b\"\nversion = \"0.1.0\"\nedition = \"2021\"\n",
        ),
        ("crates/b/src/lib.rs", "pub fn b() {}\n"),
        ("web/package.json", "{\"name\":\"web\",\"version\":\"1.0.0\"}\n"),
        ("web/index.js", "\n"),
    ];
    for (rel, body) in files {
        let path = root.join(rel);
        std::fs::create_dir_all(path.parent().unwrap()).unwrap();
        std::fs::write(path, body).unwrap();
    }
    root
}

fn counters(root: &Path) {
    let collector = PerformanceCollector::new_shared();
    let started = Instant::now();
    let found = with_current_collector(Some(Arc::clone(&collector)), || {
        detect_repo_structure(root).unwrap().is_some()
    });
    let report = collector.snapshot(started.elapsed());
    println!("root={} repo_detected={found}", root.display());
    for (name, value) in &report.counters {
        println!("{name}={value}");
    }
}

fn stats(label: &str, mut per_iter_us: Vec<f64>) {
    per_iter_us.sort_by(f64::total_cmp);
    let n = per_iter_us.len();
    let median = if n % 2 == 0 {
        (per_iter_us[n / 2 - 1] + per_iter_us[n / 2]) / 2.0
    } else {
        per_iter_us[n / 2]
    };
    let p95 = per_iter_us[((n as f64 * 0.95).ceil() as usize).saturating_sub(1).min(n - 1)];
    println!(
        "{label}: n={n} median_us={median:.1} min_us={:.1} p95_us={p95:.1} max_us={:.1}",
        per_iter_us[0],
        per_iter_us[n - 1]
    );
}

fn tiny_latency() {
    const WARMUP_ITERS: usize = 200;
    const SAMPLES: usize = 20;
    const ITERS_PER_SAMPLE: usize = 200;

    let root = tiny_tree();
    let cases: [(&str, &dyn Fn()); 3] = [
        ("serial_reference", &|| {
            std::hint::black_box(nested_benchmark::serial_reference_walk(&root));
        }),
        ("production", &|| {
            std::hint::black_box(nested_benchmark::production_walk(&root));
        }),
        ("detect_repo_structure", &|| {
            std::hint::black_box(detect_repo_structure(&root).unwrap());
        }),
    ];
    for (_, case) in &cases {
        for _ in 0..WARMUP_ITERS {
            case();
        }
    }
    let mut samples: Vec<Vec<f64>> = vec![Vec::new(); cases.len()];
    for sample in 0..SAMPLES {
        // Rotate the case order every sample so drift hits every case alike.
        for offset in 0..cases.len() {
            let index = (sample + offset) % cases.len();
            let started = Instant::now();
            for _ in 0..ITERS_PER_SAMPLE {
                (cases[index].1)();
            }
            let per_iter = started.elapsed().as_secs_f64() * 1e6 / ITERS_PER_SAMPLE as f64;
            samples[index].push(per_iter);
        }
    }
    println!(
        "tiny_tree: root={} walked_entries={} candidates={} warmup_iters={WARMUP_ITERS} samples={SAMPLES} iters_per_sample={ITERS_PER_SAMPLE}",
        root.display(),
        nested_benchmark::corpus(&root).walked_entries,
        nested_benchmark::serial_reference_walk(&root).len()
    );
    for ((label, _), values) in cases.iter().zip(samples) {
        stats(label, values);
    }
    let _ = std::fs::remove_dir_all(&root);
}

fn concurrent(target: &str, threads: usize, requests: usize) {
    let (root, owned) = if target == "tiny" {
        (tiny_tree(), true)
    } else {
        (PathBuf::from(target), false)
    };
    // Warm the tree and the code path on one thread first.
    for _ in 0..3 {
        detect_repo_structure(&root).unwrap();
    }
    let barrier = Arc::new(Barrier::new(threads + 1));
    let handles: Vec<_> = (0..threads)
        .map(|_| {
            let barrier = Arc::clone(&barrier);
            let root = root.clone();
            std::thread::spawn(move || {
                barrier.wait();
                let mut latencies = Vec::with_capacity(requests);
                for _ in 0..requests {
                    let started = Instant::now();
                    std::hint::black_box(detect_repo_structure(&root).unwrap());
                    latencies.push(started.elapsed());
                }
                latencies
            })
        })
        .collect();
    barrier.wait();
    let started = Instant::now();
    let latencies: Vec<Duration> = handles
        .into_iter()
        .flat_map(|handle| handle.join().unwrap())
        .collect();
    let wall = started.elapsed();
    let total = latencies.len();
    println!(
        "concurrent: target={target} threads={threads} requests_per_thread={requests} total_requests={total} wall_ms={:.1} throughput_rps={:.1}",
        wall.as_secs_f64() * 1e3,
        total as f64 / wall.as_secs_f64()
    );
    stats(
        "request_latency",
        latencies.iter().map(|d| d.as_secs_f64() * 1e6).collect(),
    );
    if owned {
        let _ = std::fs::remove_dir_all(&root);
    }
}

fn main() {
    let args: Vec<String> = std::env::args().skip(1).collect();
    match args.first().map(String::as_str) {
        Some("counters") => {
            if args[1] == "tiny" {
                let root = tiny_tree();
                counters(&root);
                let _ = std::fs::remove_dir_all(&root);
            } else {
                counters(Path::new(&args[1]));
            }
        }
        Some("tiny-latency") => tiny_latency(),
        Some("concurrent") => concurrent(
            &args[1],
            args[2].parse().unwrap(),
            args[3].parse().unwrap(),
        ),
        _ => {
            eprintln!("usage: repo_perf_probe counters <tiny|root> | tiny-latency | concurrent <tiny|root> <threads> <requests>");
            std::process::exit(2);
        }
    }
}
