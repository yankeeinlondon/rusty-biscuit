#!/usr/bin/env python3
"""Alternating warmed-cache timing driver for 2026-09-21-lockfile-provenance-cost.

usage: measure.py <out-evidence-dir> [set-name-substring]

Each set runs 3 discarded warmup rounds, then 20 measured rounds. Every round
runs each side once; the side order rotates by one each round, so baseline and
changed runs alternate and neither always goes first. The 1-minute load average
is recorded before and after every set; a set whose start load exceeds 16 waits
(polling every 60 s, up to 15 minutes) and is recorded as deferred if it never
drops.
"""
import hashlib, json, os, subprocess, sys, time
from datetime import datetime, timezone

BIN = "/tmp/lpc/bin"
ROOT = "/Volumes/coding/wt/rusty-biscuit/fix-sniff"
DOC = "sniff/fixes/2026-09-21-lockfile-provenance-cost/evidence/trivial.md"
WARMUP_ROUNDS, SAMPLES, IN_PROCESS_WARMUPS = 3, 20, 3
LOAD_LIMIT, POLL_S, MAX_WAIT_S = 16.0, 60, 15 * 60

def load1():
    return float(subprocess.check_output(["sysctl", "-n", "vm.loadavg"], text=True).split()[1])

def now():
    return datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")

def sha(path):
    return hashlib.sha256(open(path, "rb").read()).hexdigest()

def bench_side(tree, profile, case):
    exe = f"{BIN}/{tree}-{profile}-lpc_bench"
    def run():
        out = subprocess.run([exe, case, ROOT, str(IN_PROCESS_WARMUPS)], cwd=ROOT,
                             stdin=subprocess.DEVNULL, capture_output=True, text=True, check=True)
        return int(out.stdout.strip())
    return exe, run

def compose_side(tree, profile):
    exe = f"{BIN}/{tree}-{profile}-md"
    def run():
        start = time.perf_counter_ns()
        subprocess.run([exe, "compose", DOC, "--output", "markdown"], cwd=ROOT,
                       stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL,
                       stderr=subprocess.DEVNULL, check=True)
        return time.perf_counter_ns() - start
    return exe, run

def sets():
    for profile in ("release", "debug"):
        yield f"{profile}-parse", "in-process ns, one CargoLockVersions::parse (read + parse) after 3 in-process warmups", [
            ("baseline", "generic-parse", *bench_side("baseline", profile, "parse")),
            ("after", "typed-parse", *bench_side("changed", profile, "parse")),
        ]
        yield f"{profile}-detect", "in-process ns, one detection call on the checkout root after 3 in-process warmups", [
            ("baseline", "detect_repo_structure-always-corroborates", *bench_side("baseline", profile, "detect-structure")),
            ("after", "detect_repo_structure-declined", *bench_side("changed", profile, "detect-structure")),
            ("after", "structure-with_lockfile_provenance-requested", *bench_side("changed", profile, "detect-provenance")),
        ]
        yield f"{profile}-compose", "wall-clock ns of one `md compose <trivial.md> --output markdown` process, cwd = checkout root", [
            ("baseline", "md-compose-baseline-binary", *compose_side("baseline", profile)),
            ("after", "md-compose-changed-binary", *compose_side("changed", profile)),
        ]

def main():
    out_dir = sys.argv[1]
    only = sys.argv[2] if len(sys.argv) > 2 else ""
    for name, unit, sides in sets():
        if only not in name:
            continue
        waited, start_load = 0, load1()
        while start_load > LOAD_LIMIT and waited < MAX_WAIT_S:
            print(f"[{name}] load {start_load} > {LOAD_LIMIT}; waiting", flush=True)
            time.sleep(POLL_S); waited += POLL_S; start_load = load1()
        record = {"set": name, "unit": unit, "start": now(), "load_before": start_load,
                  "waited_s": waited}
        if start_load > LOAD_LIMIT:
            record.update(status="deferred", end=now())
            print(f"[{name}] DEFERRED at load {start_load}", flush=True)
            samples = {}
        else:
            samples = {label: [] for _, label, _, _ in sides}
            for round_index in range(WARMUP_ROUNDS + SAMPLES):
                k = round_index % len(sides)
                for _, label, _, run in sides[k:] + sides[:k]:
                    value = run()
                    if round_index >= WARMUP_ROUNDS:
                        samples[label].append(value)
            record.update(status="measured", end=now(), load_after=load1())
        for tree_dir, label, exe, _ in sides:
            doc = dict(record, side=tree_dir, case=label, binary=os.path.basename(exe),
                       binary_sha256=sha(exe), samples_ns=samples.get(label, []))
            path = os.path.join(out_dir, tree_dir, f"{name}-{label}.json")
            with open(path, "w") as fh:
                json.dump(doc, fh, indent=1)
        print(f"[{name}] {record['status']} load {record['load_before']} -> {record.get('load_after')}", flush=True)

if __name__ == "__main__":
    main()
